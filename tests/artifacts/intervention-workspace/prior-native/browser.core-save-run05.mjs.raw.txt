function gate(name) {
  const observation = preflight({
    source: manifest.source,
    state,
    minimumGiB: MIN_RUNTIME_GIB,
    explicitPath: manifest.dockerDataPath || undefined,
    probeContainer: manifest.storageProbeContainer || undefined,
    dockerRequired: manifest.runtimeKind !== "native-postgresql",
  });
  report.checks.push({ name: "preflight:" + name, ...observation });
  save();
  assert.equal(observation.pass, true);
}
async function protect(ctx) {
  ctx.setDefaultTimeout(20000);
  ctx.setDefaultNavigationTimeout(30000);
  await ctx.route("**/*", async (r) => {
    const request = r.request(),
      url = new URL(request.url());
    const login =
      url.pathname === "/api/auth/sign-in/email" && request.method() === "POST";
    const read = ["GET", "HEAD"].includes(request.method());
    const body = request.method() === "POST" ? request.postDataJSON() : null;
    const mutation =
      url.pathname === api &&
      request.method() === "POST" &&
      body?.action === allowedAction &&
      allowedAction !== null;
    const activation =
      mode === "activate" &&
      allowedAction === "activate" &&
      url.pathname === assignmentApi &&
      request.method() === "POST";
    if (url.origin === origin && (login || read || mutation || activation))
      return r.continue();
    report.blocked.push({
      url: url.origin + url.pathname,
      method: request.method(),
    });
    save();
    await r.abort("blockedbyclient");
  });
}

async function show(label) {
  const tab = page.getByRole("tab", { name: label, exact: true });
  if (await tab.isVisible()) {
    await tab.click();
    await page
      .getByRole("tab", { name: label, exact: true, selected: true })
      .waitFor();
    return;
  }
  const trigger = page.getByRole("button", {
    name: `Open ${label}`,
    exact: true,
  });
  if (await trigger.isVisible()) {
    if (
      await page.getByRole("dialog", { name: label, exact: true }).isVisible()
    )
      return;
    await trigger.click();
    return;
  }
  const modal = page.locator('dialog[aria-modal="true"][open]');
  if (await modal.count()) await page.keyboard.press("Escape");
}
async function shot(name) {
  const geometry = await page.evaluate(() => {
    const pane = document.querySelector(".tw-panes"),
      box = pane?.getBoundingClientRect();
    return {
      width: innerWidth,
      height: innerHeight,
      documentWidth: document.documentElement.scrollWidth,
      dpr: devicePixelRatio,
      scrollY,
      workspace: box
        ? { x: box.x, y: box.y, width: box.width, height: box.height }
        : null,
      focus: document.activeElement?.textContent?.trim().slice(0, 120),
    };
  });
  assert.ok(
    geometry.documentWidth <= geometry.width + 1,
    "Document horizontal overflow",
  );
  const nameWithSize = `${name}-${geometry.width}x${geometry.height}.png`;
  await page.screenshot({ path: join(out, nameWithSize), fullPage: false });
  report.screenshots.push({
    name: nameWithSize,
    url: page.url(),
    geometry,
    sha256: hash(readFileSync(join(out, nameWithSize))),
  });
  save();
}
async function captureState(name, requirePanes = true) {
  const mutationsBefore = report.mutations.length;
  for (const [width, height] of [
    [1440, 1000],
    [1024, 768],
    [390, 844],
    [320, 844],
  ]) {
    await page.setViewportSize({ width, height });
    if (requirePanes)
      await page
        .locator(
          `.tw-panes[data-layout="${width <= 800 ? "compact" : width <= 1240 ? "medium" : "wide"}"]`,
        )
        .waitFor();
    // A context URL retained from the phone may open the tablet drawer on
    // resize. Close it before testing its actual keyboard opener; never press
    // Enter with an unknown control focused inside an already open drawer.
    if (
      width === 1024 &&
      requirePanes &&
      (await page.locator('dialog[aria-modal="true"][open]').count())
    )
      await page.keyboard.press("Escape");
    await page.evaluate(() => window.scrollTo(0, 0));
    await shot(name + "-page-top");
    if (width <= 800 && requirePanes) {
      for (const label of ["Evidence", "Activity", "Steps"]) {
        const tab = page.getByRole("tab", { name: label, exact: true });
        assert.equal(await tab.count(), 1, label + " phone tab");
        await tab.click();
        await page
          .getByRole("tab", { name: label, exact: true, selected: true })
          .waitFor();
        assert.equal(await tab.getAttribute("aria-selected"), "true");
        assert.equal(await page.getByRole("tabpanel").count(), 1);
        await shot(name + "-" + label.toLowerCase());
      }
    }
    if (width === 1024 && requirePanes) {
      const trigger = page.locator(".tw-panes-context-trigger");
      assert.equal(await trigger.count(), 1);
      await trigger.focus();
      await page.keyboard.press("Enter");
      await page.getByRole("dialog").waitFor({ state: "visible" });
      assert.equal(
        await page.getByRole("dialog").evaluate((el) => el.matches(":modal")),
        true,
      );
      await trigger.evaluate((el) => el.focus());
      assert.equal(
        await trigger.evaluate((el) => el === document.activeElement),
        false,
        "Native modal makes background opener inert",
      );
      for (let n = 0; n < 8; n++) {
        await page.keyboard.press(n === 7 ? "Shift+Tab" : "Tab");
        report.checks.push({
          name: "tablet-dialog-focus-observation",
          state: name,
          index: n,
          ...(await page.evaluate(() => ({
            tag: document.activeElement?.tagName,
            id: document.activeElement?.id,
            text: document.activeElement?.textContent?.trim().slice(0, 80),
            modal: document.querySelector("dialog")?.matches(":modal"),
          }))),
        });
        save();
        if (
          !(await page
            .getByRole("dialog")
            .evaluate((el) => el.contains(document.activeElement)))
        ) {
          assert.equal(
            await page.evaluate(() => document.activeElement === document.body),
            true,
            "No background control receives keyboard focus",
          );
          assert.equal(
            await page
              .getByRole("dialog")
              .evaluate((el) => el.matches(":modal")),
            true,
          );
          // A native dialog permits browser chrome in its tab cycle. The next
          // Tab must re-enter this modal, never the underlying workspace.
          await page.keyboard.press("Tab");
          assert.equal(
            await page
              .getByRole("dialog")
              .evaluate((el) => el.contains(document.activeElement)),
            true,
            "Tab returns from browser chrome to the native modal",
          );
        }
      }
      await shot(name + "-context-open");
      await page.keyboard.press("Escape");
      assert.equal(
        await trigger.evaluate((el) => el === document.activeElement),
        true,
        "Escape returns focus to context opener",
      );
    }
  }
  assert.equal(
    report.mutations.length,
    mutationsBefore,
    "Pane browsing made a mutation",
  );
}

async function action(kind, button, validate) {
  gate(kind);
  allowedAction = kind;
  uncertain = true;
  const item = {
    action: kind,
    status: "intent-recorded",
    startedAt: new Date().toISOString(),
  };
  report.mutations.push(item);
  report.lock_retained = true;
  save();
  const responsePromise = page.waitForResponse(
    (r) =>
      new URL(r.url()).pathname === api &&
      r.request().method() === "POST" &&
      r.request().postDataJSON()?.action === kind,
    { timeout: 25000 },
  );
  await button.click();
  const response = await responsePromise;
  item.httpStatus = response.status();
  item.request = response.request().postDataJSON();
  save();
  assert.equal(
    response.status(),
    200,
    kind + " outcome requires reconciliation",
  );
  // Next may refresh after the response headers and make Playwright's page
  // response body unavailable. Reconcile using an authenticated GET only;
  // never reissue the mutation or infer success from the HTTP status alone.
  const body = await readReview();
  item.outcomeReadVia = "authenticated GET after single POST HTTP200";
  assert.equal(body.draft?.id, fixture.draft_id);
  assert.equal(
    body.draft.revision,
    item.request.expected_revision + (kind === "deliver" ? 0 : 1),
  );
  validate(body);
  const displayedStatus =
    kind === "deliver"
      ? "Delivered"
      : kind === "approve"
        ? "Approved · awaiting delivery"
        : "Private draft";
  await page.waitForFunction(
    ({ revision, status }) => {
      const displayed = document.querySelector(".lp-intervention-status");
      return (
        displayed?.querySelector("strong")?.textContent.trim() === status &&
        displayed?.querySelector("span")?.textContent.trim() ===
          `Revision ${revision} · saved`
      );
    },
    { revision: body.draft.revision, status: displayedStatus },
  );
  item.uiStatus = displayedStatus;
  item.uiRevision = body.draft.revision;
  uncertain = false;
  allowedAction = null;
  Object.assign(item, {
    status: "acknowledged",
    revision: body.draft.revision,
    approval_id: body.approval?.id ?? null,
    delivery_id: body.delivery?.id ?? null,
    finishedAt: new Date().toISOString(),
  });
  report.lock_retained = false;
  save();
  return body;
}

      afterText = await afterQuotes.allTextContents();
    assert.ok(beforeText.length && afterText.length);
    assert.ok(
      beforeText.every((text) => originals.includes(text)),
      "Rendered original text matches exact saved evidence",
    );
    assert.ok(
      afterText.every((text) => followups.includes(text)),
      "Rendered follow-up text matches exact saved response",
    );
    await page.locator(".lp-intervention-followup").evaluate((el) => {
      const body = el.closest(".tw-panes-body");
      if (body) body.scrollTop = 0;

    await shot("followup-original-expanded");
    await afterQuotes.first().scrollIntoViewIfNeeded();

      ...labels,

  assert.equal(report.pageErrors.length, 0);
